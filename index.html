btnDownloadUrl.addEventListener('click', async () => { const url = urlInput.value.trim(); if (!url) { downloadStatus.textContent = 'Pega un enlace válido.'; return; } downloadStatus.textContent = 'Buscando servidor disponible...'; // Lista ampliada de instancias públicas de extracción const apiEndpoints = [ 'https://cobaltapi.onrender.com/', 'https://api.cobalt.tools/', 'https://co.wuk.sh/api/json', 'https://cobalt-backend.canine.tools/' ]; let audioUrl = null; let filename = 'Pista Descargada'; for (let endpoint of apiEndpoints) { try { downloadStatus.textContent = 'Conectando con extractor...'; const res = await fetch(endpoint, { method: 'POST', headers: { 'Accept': 'application/json', 'Content-Type': 'application/json' }, body: JSON.stringify({ url: url, downloadMode: 'audio', audioFormat: 'm4a', youtubeVideoCodec: 'h264' }) }); if (!res.ok) continue; const data = await res.json(); if (data && (data.url || data.status === 'stream' || data.status === 'redirect')) { audioUrl = data.url; if (data.filename) { filename = data.filename.replace(/\.[^/.]+$/, ""); } break; } } catch (err) { console.warn('Servidor no disponible, probando el siguiente...', err); } } if (audioUrl) { downloadStatus.textContent = 'Descargando audio a tu memoria...'; try { // Intentar guardar como archivo físico local const fileResp = await fetch(audioUrl); const blob = await fileResp.blob(); const localBlobUrl = URL.createObjectURL(blob); const newTrack = { id: `${Date.now()}_${filename}`, index: allTracks.length, file: blob, url: localBlobUrl, title: filename, artist: 'Descarga YouTube' }; allTracks.push(newTrack); renderPlaylist(); loadTrack(newTrack.index); playAudio(); downloadStatus.textContent = '¡Completado con éxito!'; urlInput.value = ''; setTimeout(() => { downloadStatus.textContent = ''; }, 3500); } catch (fetchErr) { // Si el navegador bloquea la descarga directa del blob por CORS, se reproduce en streaming directo const newTrack = { id: `${Date.now()}_${filename}`, index: allTracks.length, file: null, url: audioUrl, title: filename, artist: 'Streaming Directo' }; allTracks.push(newTrack); renderPlaylist(); loadTrack(newTrack.index); playAudio(); downloadStatus.textContent = '¡Reproduciendo vía enlace directo!'; urlInput.value = ''; setTimeout(() => { downloadStatus.textContent = ''; }, 3500); } } else { downloadStatus.textContent = 'Servidores ocupados. Usa la alternativa rápida.'; } });
